# Esmerelda backend — production runtime dependencies.
#
# Scope: what the deployed FastAPI app actually imports at runtime —
# main.py, api/, storage/, the MCP bridge (api/mcp_bridge.py ->
# mcp-server-sqlite), and moodle/sync_service.py (POST /api/sync/moodle).
# Deliberately excludes pyttsx3 / other Windows-only voice packages —
# not imported by api/, storage/, or moodle/sync_service.py at all.
#
# Playwright *is* included below — an earlier version of this file
# excluded it because nothing in the deployed API needed it yet. That
# changed once moodle/sync_service.py (the unattended, credential-based
# Moodle sync triggered by POST /api/sync/moodle) was added; see
# BUILD_LOG.md for both that change and this one. Installing the
# `playwright` package alone is not enough to run it in a container —
# backend/Dockerfile also runs `playwright install --with-deps chromium`
# to fetch the actual browser binary and its Linux system libraries.
#
# Versions below are pinned to what is verified working in local dev
# (backend/venv), except where a package is a transitive dependency of
# another pinned package here (e.g. starlette/anyio/httpx come in via
# fastapi/groq/google-genai/mcp) — those are left unpinned so pip can
# resolve versions compatible with the pins below.
#
# Target Python: 3.12 (Render's default Python buildpack image; broad
# prebuilt-wheel coverage for every package below). Local dev was on
# 3.14, which is newer than what most of these packages publish wheels
# for yet — 3.12 is the safer, tested target for deployment.

fastapi==0.141.1
uvicorn[standard]==0.53.0
sqlalchemy==2.0.54
pydantic==2.13.5
python-dotenv==1.2.3
google-genai==2.24.0
groq==1.7.0
mcp==1.6.0
mcp-server-sqlite==2025.4.25
playwright==1.63.0
# Esmerelda's voice (backend/api/tts.py) — calls a public Kokoro TTS
# Gradio Space's API rather than scraping its webpage. No API key
# required: the Space is public.
gradio_client==2.7.1
# Required by FastAPI's UploadFile/File(...) form-data parsing — used by
# POST /api/transcribe (backend/api/stt.py) to receive the recorded
# audio Blob. Was already present transitively in local dev (pulled in by
# another dependency), which would have silently masked its absence from
# this pinned list until a production install without it broke that one
# endpoint — pinned explicitly now that this project directly depends on
# multipart parsing, not just incidentally has it installed.
python-multipart==0.0.32
# Text extraction for the Knowledge Base's document indexing/retrieval
# pipeline (storage/text_extraction.py) — PDF/DOCX/PPTX only, per that
# module's own documented scope. Each processes exactly one already-
# downloaded file at a time (same discipline as document downloading
# itself, see BUILD_LOG.md's memory-regression entry) — not a source of
# the same unbounded-buffering risk that entry fixed.
pypdf==6.19.0
python-docx==1.2.0
python-pptx==1.0.2
# pypdf's own optional dependency, for fully parsing embedded CFF Type1
# ("Type1C") font encodings when extracting text — this is what real
# production logs' repeated "WARNING pypdf._cmap: fontTools is required
# to fully parse the encoding..." was about (see BUILD_LOG.md). Confirmed
# directly: none of pypdf/python-docx/python-pptx/their own dependencies
# (lxml, Pillow, XlsxWriter, typing-extensions) require fontTools — it
# was genuinely absent from every real deployment of this file, which is
# why the warning was real and reproducible, not a fluke of one
# environment. Pinned explicitly and directly rather than relying on any
# other package to pull it in incidentally.
fonttools==4.65.0
